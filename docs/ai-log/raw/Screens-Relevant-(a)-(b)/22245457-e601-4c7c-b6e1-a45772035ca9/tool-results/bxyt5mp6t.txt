    66	const DAY_MS = 24 * 60 * 60 * 1000;
    67	const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

    69	const pad = (n) => String(n).padStart(2, "0");

    71	function toUtc(date) {
    72	  const [y, m, d] = date.split("-").map(Number);
    73	  return Date.UTC(y, m - 1, d);
    74	}
    75	
    76	function addDays(date, days) {
    77	  return new Date(toUtc(date) + days * DAY_MS).toISOString().slice(0, 10);
    78	}
    79	
    80	function daysBetween(from, to) {
    81	  return Math.round((toUtc(to) - toUtc(from)) / DAY_MS);
    82	}

    84	// The next payday strictly after today. Payday 28 exists in every month.
    85	function nextPayday(payday, today) {
    86	  const [y, m, d] = today.split("-").map(Number);
    87	  const [year, month] = d < payday ? [y, m] : m === 12 ? [y + 1, 1] : [y, m + 1];
    88	  return `${year}-${pad(month)}-${pad(payday)}`;
    89	}

    93	function feeFor(amount) {
    94	  const fee = (amount * FEE_PERCENT) / 100;
    95	  if (!Number.isInteger(fee)) throw new Error(`Fee for ${amount} is not whole rupees`);
    96	  return fee;
    97	}

    99	// Only events up to and including today count.
   100	const upTo = (events, today) => events.filter((e) => e.date <= today);
   101	
   102	function paidBy(advance, today) {
   103	  return upTo(advance.payments, today).reduce((sum, p) => sum + p.amount, 0);
   104	}
   105	
   106	function repaidOn(advance, today) {
   107	  let paid = 0;
   108	  for (const p of upTo(advance.payments, today)) {
   109	    paid += p.amount;
   110	    if (paid >= advance.total) return p.date;
   111	  }
   112	  return null;
   113	}

   115	function failedAttemptOn(advance, date, today) {
   116	  return (
   117	    upTo(advance.collection_attempts, today).find(
   118	      (a) => a.date === date && a.result === "not_collected"
   119	    ) ?? null
   120	  );
   121	}

   125	export function getOffer(persona, today) {
   126	  if (!persona.income_verified || persona.limit == null || persona.payday == null) {
   127	    return {
   128	      eligibility: "needs_verification",
   129	      limit: null,
   130	      due_date: null,
   131	      days_until_due: null,
   132	      collection_time: null,
   133	      tiers: null,
   134	    };
   135	  }
   136	  const due_date = nextPayday(persona.payday, today);
   137	  return {
   138	    eligibility: "eligible",
   139	    limit: persona.limit,
   140	    due_date,
   141	    days_until_due: daysBetween(today, due_date),
   142	    collection_time: COLLECTION_TIME, // so the offer screen can say "at 6pm" without reaching for a constant
   143	    tiers: TIERS.map((amount) =>
   144	      amount <= persona.limit
   145	        ? { amount, fee: feeFor(amount), total: amount + feeFor(amount), available: true, blocked_reason: null }
   146	        : { amount, fee: null, total: null, available: false, blocked_reason: "over_limit" }
   147	    ),
   148	  };
   149	}

   151	function buildTerms({ persona_id, offered_on, applied_amount, disbursed_amount, due_date, reduced_from }) {
   152	  const fee = feeFor(disbursed_amount);
   153	  const total = disbursed_amount + fee;
   154	  const default_date = addDays(due_date, DEFAULT_AFTER_DAYS);
   155	  return {
   156	    id: `${persona_id}-${offered_on}-${applied_amount}-${disbursed_amount}`,
   157	    persona_id,
   158	    offered_on,
   159	    applied_amount,
   160	    disbursed_amount,
   161	    fee,
   162	    total,
   163	    due_date,
   164	    collection: { date: due_date, time: COLLECTION_TIME, from: "novapay_wallet" },
   165	    default_date,
   166	    // MR: the permission ends when repaid or at the default date, whichever comes first.
   167	    permission: { amount: total, from_date: due_date, ends_by: default_date },
   168	    reduced_from,
   169	  };
   170	}

   183	    reduced_from: null,

   229	// MR: collection is full amount only, from the NovaPay wallet, once salary has landed.
   230	// After payday, it's attempted again only when salary lands (decided, not any incoming money).
   231	export function collectionOutcome(advance, { date, time, salary_landed, salary_amount = null, balance, salary_destination = "novapay" }) {
   232	  const owed = advance.total - paidBy(advance, date);
   233	  if (owed <= 0) return advance;
   234	  let reason = null;
   235	  if (salary_destination !== "novapay") reason = "salary_moved_to_bank";
   236	  else if (!salary_landed) reason = "salary_not_arrived";
   237	  else if (balance < owed && salary_amount < owed) reason = "salary_short";
   238	  else if (balance < owed) reason = "balance_dropped";
   239	  return {
   240	    ...advance,
   241	    collection_attempts: [
   242	      ...advance.collection_attempts,
   243	      { date, time, result: reason ? "not_collected" : "collected", reason },
   244	    ],
   245	    payments: reason ? advance.payments : [...advance.payments, { date, amount: owed, method: "auto" }],
   246	  };
   247	}

   254	export function getStatus(advance, today) {
   255	  const repaid_on = repaidOn(advance, today);
   256	  const attempts = upTo(advance.collection_attempts, today);
   257	  const reported = advance.payroll_delay_reported_on;
   258	  const common = {
   259	    outstanding: advance.total - paidBy(advance, today),
   260	    repaid_on,
   261	    permission: {
   262	      active: repaid_on == null && today < advance.default_date,
   263	      ends_on: repaid_on != null && repaid_on < advance.default_date ? repaid_on : advance.default_date,
   264	    },
   265	    last_attempt: attempts.length ? attempts[attempts.length - 1] : null,
   266	    reminders_paused: reported != null && reported <= today,
   267	  };
   268	
   269	  // Days late are counted from the due date and shown truthfully. No grace rule.
   270	  if (repaid_on) {
   271	    const days_late = Math.max(0, daysBetween(advance.due_date, repaid_on));
   272	    const repayment_status = days_late === 0 ? "on_time" : days_late >= DEFAULT_AFTER_DAYS ? "default" : "late";
   273	    return { phase: "repaid", repayment_status, days_late, days_until_due: null, days_until_default: null, ...common };
   274	  }
   275	
   276	  const sinceDue = daysBetween(advance.due_date, today);
   277	  const days_late = Math.max(0, sinceDue);
   278	  let phase;
   279	  if (sinceDue < 0) phase = "before_due";
   280	  else if (sinceDue === 0) phase = failedAttemptOn(advance, advance.due_date, today) ? "not_collected" : "due_today";
   281	  else if (days_late >= DEFAULT_AFTER_DAYS) phase = "default";
   282	  else phase = "late";
   283	
   284	  return {
   285	    phase,
   286	    repayment_status: phase === "late" || phase === "default" ? phase : null,
   287	    days_late,
   288	    days_until_due: sinceDue <= 0 ? -sinceDue : null,
   289	    days_until_default: phase === "default" ? null : daysBetween(today, advance.default_date),
   290	    ...common,
   291	  };
   292	}

   296	// True if adding `date` keeps every 7-day window at or under the cap.
   297	function fitsCap(kept, date) {
   298	  for (let start = addDays(date, -6); start <= date; start = addDays(start, 1)) {
   299	    const end = addDays(start, 6);
   300	    const inWindow = kept.filter((r) => r.date >= start && r.date <= end).length;
   301	    if (inWindow >= MAX_REMINDERS_PER_7_DAYS) return false;
   302	  }
   303	  return true;
   304	}

   306	// Decided: a material consequence must always be announced, so this is never paused.
   307	const NEVER_PAUSED = ["before_default"];
   308	
   309	/**
   310	 * Every reminder for this advance as known today, each with a status:
   311	 * "sent" (on or before today), "scheduled" (later, if nothing changes) or "paused".
   312	 * Notices about events win over routine late reminders when the 2-a-week cap bites.
   313	 * Nothing is sent on or after the default date.
   314	 */
   315	export function getReminders(advance, today) {
   316	  const repaid_on = repaidOn(advance, today);
   317	  const owedOn = (date) => repaid_on == null || repaid_on >= date;
   318	  const due = advance.due_date;
   319	  const failed = failedAttemptOn(advance, due, today);
   320	  const lateKnown = failed != null || (today > due && owedOn(due));
   321	  const candidates = [];
   322	
   323	  const beforeDue = addDays(due, -REMINDER_DAYS_BEFORE_DUE);
   324	  if (beforeDue >= advance.accepted_on && owedOn(beforeDue)) {
   325	    candidates.push({ type: "before_due", date: beforeDue, time: REMINDER_TIME, priority: 1 });
   326	  }
   327	  if (failed) {
   328	    // Sent when the collection fails, not at 10:00: the attempt is at 6pm. Inside 8am–10pm.
   329	    candidates.push({ type: "not_collected", date: due, time: failed.time, priority: 0 });
   330	  }
   331	  if (lateKnown) {
   332	    for (let d = addDays(due, LATE_REMINDER_EVERY_DAYS); d < advance.default_date; d = addDays(d, LATE_REMINDER_EVERY_DAYS)) {
   333	      if (owedOn(d)) candidates.push({ type: "late", date: d, time: REMINDER_TIME, priority: 2 });
   334	    }
   335	    const beforeDefault = addDays(advance.default_date, -NOTICE_DAYS_BEFORE_DEFAULT);
   336	    if (owedOn(beforeDefault)) {
   337	      candidates.push({ type: "before_default", date: beforeDefault, time: REMINDER_TIME, priority: 0 });
   338	    }
   339	  }
   340	
   341	  const reported = advance.payroll_delay_reported_on;
   342	  const pausedAfter = reported != null && reported <= today ? reported : null;
   343	  const kept = [];
   344	  const out = [];
   345	  candidates.sort((a, b) => a.priority - b.priority || a.date.localeCompare(b.date));
   346	  for (const c of candidates) {
   347	    if (pausedAfter && c.date > pausedAfter && !NEVER_PAUSED.includes(c.type)) {
   348	      out.push({ ...c, status: "paused" });
   349	    } else if (fitsCap(kept, c.date)) {
   350	      kept.push(c);
   351	      out.push({ ...c, status: c.date <= today ? "sent" : "scheduled" });
   352	    }
   353	  }
   354	  return out
   355	    .sort((a, b) => a.date.localeCompare(b.date) || a.time.localeCompare(b.time))
   356	    .map(({ priority, ...reminder }) => reminder);
   357	}

     3	/**
     4	 * Choice — NovaKit Lite
     5	 * One component for a checkbox or a radio: a native <input> inside a <label>.
     6	 * Types: checkbox | radio
     7	 * States: unchecked, checked, focus-visible, disabled (with a visible reason)
     8	 *
     9	 * - Controlled only. It never defaults to checked: pass `checked` and `onChange`.
    10	 * - The whole row is the tap target, at least 44px tall.
    11	 * - The accessible name is the label only. The description and the disabled reason
    12	 *   are read as descriptions (aria-describedby), so they're heard but not repeated.
    13	 * - Disabled uses aria-disabled, not the native attribute, so the option stays
    14	 *   focusable and its reason is heard. It can't be checked: a tap is cancelled and
    15	 *   onChange is never called. A caller's onClick still runs, so a screen can say why.
    16	 *   The control is dimmed; the label and reason stay readable in neutral-700.
    17	 * - Any other props (aria-*, required, form, onClick…) go to the <input>.
    18	 *
    19	 * Radio groups: the caller wraps the options in a <fieldset> with a <legend>
    20	 * that names the choice, and gives every option the same `name`. For example:
    21	 *
    22	 *   <fieldset>
    23	 *     <legend>How much do you need?</legend>
    24	 *     <Choice type="radio" name="amount" value="5000" checked={…} onChange={…} label="Rs 5,000" />
    25	 *     <Choice type="radio" name="amount" value="10000" checked={…} onChange={…} label="Rs 10,000" />
    26	 *   </fieldset>
    27	 */
    28	export default function Choice({
    29	  type = "checkbox",
    30	  name,
    31	  value,
    32	  checked = false,
    33	  onChange,
    34	  label,
    35	  description = null,
    36	  disabled = false,
    37	  disabledReason = null,
    38	  id,
    39	  onClick,
    40	  ...rest
    41	}) {
    42	  const autoId = useId();
    43	  const inputId = id ?? `choice-${autoId}`;
    44	  const labelId = `${inputId}-label`;
    45	  const descriptionId = description ? `${inputId}-description` : null;
    46	  const reasonId = disabled && disabledReason ? `${inputId}-reason` : null;
    47	  const describedBy =
    48	    [rest["aria-describedby"], descriptionId, reasonId].filter(Boolean).join(" ") || undefined;
    49	
    50	  function handleClick(event) {
    51	    if (disabled) event.preventDefault(); // cancels the check, by tap, click or Space
    52	    onClick?.(event);
    53	  }
    54	
    55	  return (
    56	    <label
    57	      htmlFor={inputId}
    58	      className={

=== Button
     1	/**
     2	 * Button — NovaKit Lite
     3	 * Variants: primary | secondary
     4	 * Sizes: md | lg (minimum heights, so a second line fits)
     5	 * States: default, pressed (active), focus-visible, disabled, loading
     6	 *
     7	 * - subLabel: an optional second line, e.g. "Repay Rs 10,300 on 28 Jul". Screen readers
     8	 *   hear a stop between the lines: "Accept Rs 10,000. Repay Rs 10,300 on 28 Jul".
     9	 * - loading: the button is disabled and aria-busy. loadingLabel, if given, replaces
    10	 *   the label; the original label stays in place invisibly, so the size never changes.
    11	 * - Any other props (aria-*, id, form…) go to the <button>.
    12	 */
    13	export default function Button({
    14	  children,
    15	  variant = "primary",
    16	  size = "md",
    17	  disabled = false,
    18	  loading = false,
    19	  loadingLabel = null,
    20	  subLabel = null,
    21	  onClick,
    22	  type = "button",
    23	  className = "",
    24	  ...rest
    25	}) {
    26	  const base =
    27	    "inline-flex flex-col items-center justify-center text-center rounded-md font-semibold transition-colors select-none w-full py-2 " +
    28	    "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand focus-visible:ring-offset-2";
    29	
    30	  const sizes = {
    31	    md: "min-h-11 px-4 text-body",
    32	    lg: "min-h-14 px-5 text-title",
    33	  };
    34	
    35	  // While loading, the button keeps its own colours: it's working, not unavailable.
    36	  const variants = {
    37	    primary: {
    38	      enabled: "bg-brand text-white active:bg-brand-pressed",
    39	      disabled: "bg-neutral-300 text-neutral-500",
    40	      loading: "bg-brand text-white cursor-progress",
    41	      subLabel: "text-brand-100",
    42	    },
    43	    secondary: {
    44	      enabled: "bg-white text-neutral-900 border border-neutral-300 active:bg-neutral-100",
    45	      disabled: "bg-white text-neutral-300 border border-neutral-300",
    46	      loading: "bg-white text-neutral-900 border border-neutral-300 cursor-progress",
    47	      subLabel: "text-neutral-700",
    48	    },
    49	  };
    50	  const v = variants[variant];
    51	  const state = loading ? "loading" : disabled ? "disabled" : "enabled";
    52	  const showLoadingLabel = loading && loadingLabel != null;
    53	
    54	  return (
    55	    <button
    56	      {...rest}
    57	      type={type}
    58	      disabled={disabled || loading}
    59	      aria-busy={loading || undefined}
    60	      onClick={onClick}
    61	      className={`${base} ${sizes[size]} ${v[state]} ${className}`}
    62	    >
    63	      <span className="grid">
    64	        <span className={`col-start-1 row-start-1 ${showLoadingLabel ? "invisible" : ""}`}>{children}</span>
    65	        {showLoadingLabel ? <span className="col-start-1 row-start-1">{loadingLabel}</span> : null}
    66	        {/* A stop heard but not seen, so the two lines aren't read as one phrase. */}
    67	        {subLabel ? <span className="sr-only">.</span> : null}
    68	      </span>
    69	      {subLabel ? (
    70	        <span className={`text-caption ${state === "disabled" ? "" : v.subLabel}`}>{subLabel}</span>
    71	      ) : null}
    72	    </button>
    73	  );
    74	}

     1	/**
